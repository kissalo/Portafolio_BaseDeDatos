CREATE DATABASE sistemabiblioteca;
USE sistemabiblioteca;
CREATE TABLE Libro (
código varchar(5) PRIMARY KEY
);

CREATE TABLE Autor (
código varchar(5) PRIMARY KEY
);

CREATE TABLE Libro_Autor (
libro varchar (5) NOT NULL,
autor varchar (5) NOT NULL,
PRIMARY KEY (libro, autor),
FOREIGN KEY (libro) 
REFERENCES Libro (código),
FOREIGN KEY (autor) 
REFERENCES Autor (codigo)
);

CREATE TABLE Socio (
código varchar(5) PRIMARY KEY
);

CREATE TABLE Prestamo (
código varchar (5) PRIMARY KEY,
fechaRetiro Date NOT NULL,
fechaDevolucion Date NOT NULL,
libro varchar (5) NOT NULL,
socio varchar (5) NOT NULL,
FOREIGN KEY (libro) 
REFERENCES Libro (código)
);





CREATE DATABASE pedidos_restaurante;
USE pedidos_restaurante;
CREATE TABLE Cliente (
	código varchar (10) PRIMARY KEY
	);

CREATE TABLE Pedido (
	código varchar (5) PRIMARY KEY,
	precio double NOT NULL,
	cliente varchar (10) NOT NULL UNIQUE,
	FOREIGN KEY (cliente)
	REFERENCES Cliente (código)
	);

CREATE TABLE Plato (
	código varchar (5) PRIMARY KEY,
	categoria varchar (15) NOT NULL,
	pedido varchar (5) NOT NULL,
	FOREIGN KEY (pedido)
	REFERENCES Pedido (código)
	);






CREATE DATABASE clinica_medica;
USE clinica_medica;
CREATE TABLE Especialidad (
	código varchar (15) PRIMARY KEY
	);

CREATE TABLE Medico (
	código varchar (10) PRIMARY KEY,
	especialidad varchar (15) NOT NULL,
	FOREIGN KEY (especialidad)
	REFERENCES Especialidad (código)
	);

CREATE TABLE Paciente (
	código varchar (10) PRIMARY KEY
	);

CREATE TABLE Cita (
	código varchar (5) PRIMARY KEY,
	fecha Date NOT NULL,
	hora DateTime NOT NULL,
	consultorio varchar (6) NOT NULL,
	paciente varchar (10) NOT NULL,
	medico varchar (10) NOT NULL,
	FOREIGN KEY (paciente) 
	REFERENCES Paciente (código),
	FOREIGN KEY (medico) 
	REFERENCES Medico (código)
	);

CREATE TABLE Receta (
	código varchar (7) PRIMARY KEY,
	dosis text NOT NULL,
	duración int NOT NULL,
	cita varchar (5) NOT NULL,
	FOREIGN KEY (cita) 
	REFERENCES Cita (código)
	);
	
CREATE TABLE Medicamento (
	código varchar (7) PRIMARY KEY,
	receta varchar (7) NOT NULL,
	FOREIGN KEY (receta) 
	REFERENCES Receta (código)
	);

CREATE DATABASE tienda_informatica;
USE tienda_informatica;
CREATE TABLE Proveedor (
	código varchar (10) PRIMARY KEY,
	nombres varchar (50) NOT NULL,
	apellidos varchar (50) NOT NULL,
	dirección text,
	numeroTelefonico varchar (10) NOT NULL,
	provincia varchar (35) NOT NULL
	);

CREATE TABLE Cliente (
	código varchar (10) PRIMARY KEY,
	nombres varchar (50) NOT NULL,
	apellidos varchar (50) NOT NULL,
	dirección text,
	numeroTelefonico varchar (10) NOT NULL
	);

CREATE TABLE Producto (
	código varchar (7) PRIMARY KEY,
	descripción text,
	precio double NOT NULL
	);

CREATE TABLE Compra (
	identificador varchar (9) PRIMARY KEY,
	fecha Date NOT NULL	
	);


CREATE TABLE Producto_Compra (
	producto varchar (7) NOT NULL,
	compra varchar (9) NOT NULL,
	PRIMARY KEY (producto, compra),
	FOREIGN KEY (producto)
	REFERENCES Producto (código),
	FOREIGN KEY (compra)
	REFERENCES Compra (identificador)
	);

CREATE TABLE Cliente_Compra (
	cliente varchar (7) NOT NULL,
	compra varchar (9) NOT NULL,
	PRIMARY KEY (cliente, compra),
	FOREIGN KEY (compra)
	REFERENCES Compra (identificador),
	FOREIGN KEY (cliente)
	REFERENCES Cliente (código)
	);

CREATE TABLE Proveedor_Producto (
	producto varchar (7) NOT NULL,
	proveedor varchar (10) NOT NULL,
	PRIMARY KEY (producto, proveedor),
	FOREIGN KEY (producto)
	REFERENCES Producto (código),
	FOREIGN KEY (proveedor)
	REFERENCES Proveedor (código)
	);












CREATE DATABASE empresa_discografica;
USE empresa_discografica;
CREATE TABLE Cantante (
	identificador varchar (10) PRIMARY KEY,
	nombre varchar (50) NOT NULL,
	país varchar (35) NOT NULL
	);

CREATE TABLE GeneroMusical (
	identificador varchar (25) PRIMARY KEY
	);

CREATE TABLE Disco (
	identificador varchar (7) PRIMARY KEY,
	titulo varchar (35) NOT NULL,
	precio double NOT NULL,
	generoMusical varchar (25) NOT NULL,
	cantante varchar (10) NOT NULL,
	FOREIGN KEY (cantante)
	REFERENCES Cantante (identificador) 
	);

CREATE TABLE Cancion (
	identificador varchar (15) PRIMARY KEY,
	titulo varchar (35) NOT NULL,
	disco varchar (7) NOT NULL,
	FOREIGN KEY (disco)
	REFERENCES Disco (identificador) 
	);

CREATE TABLE Disco_Cancion (
	cancion varchar (15),
	disco varchar (7),
	PRIMARY KEY (cancion, disco),
	FOREIGN KEY (disco)
	REFERENCES Disco (identificador),
	FOREIGN KEY (cancion)
	REFERENCES Cancion (identificador)
	);











CREATE DATABASE camiones;
USE camiones;
CREATE TABLE Camionero (
	cedula varchar (10) PRIMARY KEY,
	telefono varchar (10) NOT NULL,
	nombre varchar (50) NOT NULL,
	direcccion text,
	salario double NOT NULL,
	poblacion varchar (35) NOT NULL
	);

CREATE TABLE Camion (
	matricula varchar (8) PRIMARY KEY,
	modelo varchar (25) NOT NULL,
	tipo varchar (25) NOT NULL,
	potencia double NOT NULL
	);

CREATE TABLE Provincia (
	codigo varchar (35) PRIMARY KEY,
	nombre varchar (50) NOT NULL
	);

CREATE TABLE Paquete (
	codigo varchar (7) PRIMARY KEY,
	destinatario varchar (15) NOT NULL,
	direccionDestinatario text NOT NULL,
	descripcion text,
	provincia varchar (35) NOT NULL,
	camionero varchar (10) NOT NULL,
	FOREIGN KEY (provincia)
	REFERENCES Provincia (codigo),
	FOREIGN KEY (camionero)
	REFERENCES Camionero (cedula)	
	);

CREATE TABLE Camionero_Camion (
	camionero varchar (10) NOT NULL,
	camion varchar (8) NOT NULL,
	FOREIGN KEY (camion)
	REFERENCES Camion (matricula),
	FOREIGN KEY (camionero)
	REFERENCES Camionero (cedula)
	);
